
-- ==========================================
-- MOVIE RECOMMENDATION DATA ENGINE
-- MySQL 8.0+
-- ==========================================

DROP DATABASE IF EXISTS movie_engine;
CREATE DATABASE movie_engine
    CHARACTER SET utf8mb4
    COLLATE utf8mb4_unicode_ci;

USE movie_engine;

-- 1. USERS TABLE
CREATE TABLE users (
    user_id INT AUTO_INCREMENT PRIMARY KEY,
    user_name VARCHAR(100) NOT NULL,
    favorite_genre_id INT
);

-- 2. GENRES TABLE
CREATE TABLE genres (
    genre_id INT AUTO_INCREMENT PRIMARY KEY,
    genre_name VARCHAR(50) NOT NULL UNIQUE
);

-- 3. MOVIES TABLE
CREATE TABLE movies (
    movie_id INT AUTO_INCREMENT PRIMARY KEY,
    movie_title VARCHAR(150) NOT NULL,
    genre_id INT NOT NULL,
    release_year YEAR NOT NULL,
    FOREIGN KEY (genre_id)
        REFERENCES genres(genre_id)
);

-- 4. ADD USER-GENRE RELATIONSHIP
ALTER TABLE users
ADD CONSTRAINT fk_users_favorite_genre
FOREIGN KEY (favorite_genre_id)
REFERENCES genres(genre_id)
ON DELETE SET NULL;

-- 5. RATINGS TABLE
CREATE TABLE ratings (
    rating_id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    movie_id INT NOT NULL,
    rating DECIMAL(2,1) NOT NULL,
    rated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT chk_rating_range
        CHECK (rating >= 1.0 AND rating <= 5.0),

    CONSTRAINT uq_user_movie
        UNIQUE (user_id, movie_id),

    FOREIGN KEY (user_id)
        REFERENCES users(user_id)
        ON DELETE CASCADE,

    FOREIGN KEY (movie_id)
        REFERENCES movies(movie_id)
        ON DELETE CASCADE
);

-- ==========================================
-- QUERY 1: RECOMMEND MOVIES BY GENRE
-- ==========================================

SELECT u.user_name,
       m.movie_title,
       g.genre_name,
       m.release_year
FROM users u
JOIN genres g
    ON g.genre_id = u.favorite_genre_id
JOIN movies m
    ON m.genre_id = g.genre_id
LEFT JOIN ratings r
    ON r.user_id = u.user_id
   AND r.movie_id = m.movie_id
WHERE r.rating_id IS NULL
ORDER BY u.user_name, m.movie_title;

-- ==========================================
-- QUERY 2: TOP-RATED MOVIES
-- ==========================================

SELECT m.movie_title,
       ROUND(AVG(r.rating), 2) AS average_rating,
       COUNT(*) AS total_ratings
FROM movies m
JOIN ratings r
    ON r.movie_id = m.movie_id
GROUP BY m.movie_id, m.movie_title
HAVING COUNT(*) >= 2
ORDER BY average_rating DESC,
         total_ratings DESC;

-- ==========================================
-- QUERY 3: SIMILAR MOVIES BY GENRE
-- ==========================================

SELECT g.genre_name,
       m1.movie_title AS movie_one,
       m2.movie_title AS movie_two
FROM movies m1
JOIN movies m2
    ON m1.genre_id = m2.genre_id
   AND m1.movie_id < m2.movie_id
JOIN genres g
    ON g.genre_id = m1.genre_id
ORDER BY g.genre_name, movie_one, movie_two;

-- ==========================================
-- QUERY 4: MOST POPULAR GENRE
-- ==========================================

WITH genre_counts AS (
    SELECT g.genre_id,
           g.genre_name,
           COUNT(r.rating_id) AS total_ratings
    FROM genres g
    LEFT JOIN movies m
        ON m.genre_id = g.genre_id
    LEFT JOIN ratings r
        ON r.movie_id = m.movie_id
    GROUP BY g.genre_id, g.genre_name
)
SELECT genre_name, total_ratings
FROM genre_counts
WHERE total_ratings = (
    SELECT MAX(total_ratings)
    FROM genre_counts
)
ORDER BY genre_name;

-- ==========================================
-- QUERY 5: USER RATING ANALYSIS
-- ==========================================

SELECT u.user_name,
       ROUND(AVG(r.rating), 2) AS average_rating,
       COUNT(r.rating_id) AS movies_rated
FROM users u
LEFT JOIN ratings r
    ON r.user_id = u.user_id
GROUP BY u.user_id, u.user_name
ORDER BY movies_rated DESC, u.user_name;

-- ==========================================
-- QUERY 6: MOVIES WITH RATINGS >= 4
-- ==========================================

SELECT DISTINCT m.movie_title,
       g.genre_name,
       m.release_year
FROM movies m
JOIN genres g
    ON g.genre_id = m.genre_id
JOIN ratings r
    ON r.movie_id = m.movie_id
WHERE r.rating >= 4.0
ORDER BY m.movie_title;

-- ==========================================
-- QUERY 7: MOVIES RAHUL HAS NOT RATED
-- ==========================================

SELECT m.movie_title,
       g.genre_name,
       m.release_year
FROM movies m
JOIN genres g
    ON g.genre_id = m.genre_id
WHERE NOT EXISTS (
    SELECT 1
    FROM ratings r
    JOIN users u
        ON u.user_id = r.user_id
    WHERE u.user_name = 'Rahul'
      AND r.movie_id = m.movie_id
)
ORDER BY m.movie_title;

-- ==========================================
-- QUERY 8: GENRE-WISE ANALYSIS
-- ==========================================

SELECT g.genre_name,
       COUNT(DISTINCT m.movie_id) AS movie_count,
       ROUND(AVG(r.rating), 2) AS average_rating
FROM genres g
LEFT JOIN movies m
    ON m.genre_id = g.genre_id
LEFT JOIN ratings r
    ON r.movie_id = m.movie_id
GROUP BY g.genre_id, g.genre_name
ORDER BY movie_count DESC, g.genre_name;

-- ==========================================
-- QUERY 9: TOP 3 MOVIES
-- ==========================================

SELECT m.movie_title,
       ROUND(AVG(r.rating), 2) AS average_rating,
       COUNT(r.rating_id) AS total_ratings
FROM movies m
JOIN ratings r
    ON r.movie_id = m.movie_id
GROUP BY m.movie_id, m.movie_title
ORDER BY average_rating DESC,
         total_ratings DESC,
         m.movie_title
LIMIT 3;

JOIN movies m

## Problem 4: Find the highest-rated movie

Problem statement: Find the movie or movies with the highest average user rating.

SQL query

```

SELECT m.movie_title,
       ROUND(AVG(r.rating), 2) AS average_rating
FROM movies m
JOIN ratings r
    ON m.movie_id = r.movie_id
GROUP BY m.movie_id, m.movie_title
HAVING AVG(r.rating) = (
    SELECT MAX(avg_rating)
    FROM (
        SELECT AVG(rating) AS avg_rating
        FROM ratings
        GROUP BY movie_id
    ) AS movie_averages
);

```

Expected output

| movie_title | average_rating |
| ----------- | -------------- |
| Hi Nanna    | 5.00           |
| Jersey      | 5.00           |
| RRR         | 5.00           |
